cmake_minimum_required(VERSION 3.20)
project(FugScope VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This CMake build targets Windows x64. For macOS use scripts/build-macos.sh.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions($<$<COMPILE_LANGUAGE:CXX>:NOMINMAX> _CRT_SECURE_NO_WARNINGS)
if(MINGW)
  add_link_options(-static -static-libgcc -static-libstdc++)
endif()
add_library(glew STATIC vendor/glew/src/glew.c)
target_include_directories(glew PUBLIC vendor/glew/include)
target_compile_definitions(glew PUBLIC GLEW_STATIC GLEW_NO_GLU)
target_link_libraries(glew PUBLIC opengl32)
set(PA vendor/portaudio)
file(GLOB PA_COMMON CONFIGURE_DEPENDS ${PA}/src/common/*.c)
add_library(portaudio STATIC ${PA_COMMON}
  ${PA}/src/os/win/pa_win_hostapis.c
  ${PA}/src/os/win/pa_win_util.c
  ${PA}/src/os/win/pa_win_waveformat.c
  ${PA}/src/os/win/pa_win_wdmks_utils.c
  ${PA}/src/os/win/pa_win_coinitialize.c
  ${PA}/src/hostapi/wasapi/pa_win_wasapi.c)
if(MSVC)
  target_sources(portaudio PRIVATE ${PA}/src/os/win/pa_x86_plain_converters.c)
endif()
target_include_directories(portaudio PUBLIC ${PA}/include PRIVATE ${PA}/src/common ${PA}/src/os/win)
target_compile_definitions(portaudio PRIVATE PA_USE_WASAPI=1 PA_LITTLE_ENDIAN=1 SIZEOF_LONG=4 _WIN32_WINNT=0x0601 WINVER=0x0601 UNICODE _UNICODE)
target_link_libraries(portaudio PUBLIC winmm ole32 uuid)
add_library(FugScope MODULE src/FugScope.cpp src/AudioCapture.cpp vendor/ffgl/lib/FFGLSDK.cpp)
target_include_directories(FugScope PRIVATE src vendor/ffgl/lib)
target_link_libraries(FugScope PRIVATE portaudio glew)
set_target_properties(FugScope PROPERTIES PREFIX "" OUTPUT_NAME "FugScope")
include(CTest)
if(BUILD_TESTING)
  add_executable(core-test tests/core_test.cpp)
  target_include_directories(core-test PRIVATE src)
  add_executable(audio-test tests/audio_test.cpp src/AudioCapture.cpp)
  target_include_directories(audio-test PRIVATE src ${PA}/include)
  add_executable(windows-load-test tests/windows_load_test.cpp)
  target_include_directories(windows-load-test PRIVATE vendor/ffgl/lib vendor/glew/include)
  foreach(t core-test audio-test windows-load-test)
    if(MSVC)
      target_compile_options(${t} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${t} PRIVATE -UNDEBUG)
    endif()
  endforeach()
  add_test(NAME core COMMAND core-test)
  add_test(NAME audio COMMAND audio-test)
  add_test(NAME windows-load COMMAND windows-load-test $<TARGET_FILE:FugScope>)
endif()
install(TARGETS FugScope LIBRARY DESTINATION . RUNTIME DESTINATION .)
install(FILES LICENSE NOTICE.md README.md DESTINATION .)
install(DIRECTORY licenses DESTINATION .)
